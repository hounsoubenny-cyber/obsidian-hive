## Exemple d'Intégration Slack (Quick Win)

```python
# Dans reaction_module.py - React.block()

def send_slack_alert(ip, score, action, geoloc):
    webhook_url = os.getenv("SLACK_WEBHOOK_URL")
    if not webhook_url:
        return

    color = "#d32f2f" if score > 200 else "#f57c00" if score > 100 else "#4caf50"

    message = {
        "attachments": [{
            "color": color,
            "title": f"🚨 ShieldIA Alert - {action.upper()}",
            "fields": [
                {"title": "IP", "value": ip, "short": True},
                {"title": "Score", "value": f"{score}/300", "short": True},
                {"title": "Pays", "value": f":flag_{geoloc.lower()}: {geoloc}", "short": True},
                {"title": "Action", "value": action, "short": True}
            ],
            "actions": [
                {"type": "button", "text": "🔓 Débloquer", "url": f"http://shieldia:8080/api/unlock?ip={ip}"},
                {"type": "button", "text": "✅ Whitelist", "url": f"http://shieldia:8080/api/whitelist?ip={ip}"},
            ]
        }]
    }
    requests.post(webhook_url, json=message)
```
